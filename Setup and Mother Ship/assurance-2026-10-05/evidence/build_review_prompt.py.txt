"""Assemble a fresh-context, read-only review prompt for one Grok reviewer seat.

Usage: python build_review_prompt.py <wave> <seat_index> <out_path>
All material is supplied inline (plan contract, gate artifact, the wave's test
files, the product diff against the worktree's base commit, and the provider
modules) so the reviewer needs no shell command. Read-only: the base files come
from `git show` in the main repository; nothing in the worktree is modified.
"""
import difflib, json, re, subprocess, sys
from pathlib import Path

WT = Path(r"<worktree>")
MAIN = r"<main checkout>"
BASE = "f77710c"
PLAN = WT / "Setup and Mother Ship" / "assurance-2026-10-05" / "IMPLEMENTATION-PLAN.md"
PRODUCT = ["provider_setup.py", "provider_native.py", "provider_secrets.py", "provider_settings.py", "anchor_settings.py",
           "static/provider-setup.js", "static/provider-controls.js", "high_seat_page.py", "route_table.py", "job_runner.py",
           "steward_cockpit/steward_engine.py", "usage_capacity.py", "usage_capture.py", "report_viewer.py", "anchor_gui.py",
           "steward_cockpit/steward_routes.py"]
INLINE_FULL = {1: ["provider_setup.py", "provider_native.py"], 2: ["provider_setup.py", "provider_native.py"]}
TESTS = {1: ["tests/test_provider_setup_rigor.py"], 2: ["tests/test_provider_setup_hardening.py", "tests/test_provider_setup_rigor.py"]}
SCHEMA = {"type": "object", "required": ["answerable", "findings"], "properties": {
    "answerable": {"enum": ["yes", "no"]}, "note": {"type": "string"},
    "findings": {"type": "array", "items": {"type": "object", "required": ["severity", "file", "rule"], "properties": {
        "severity": {"enum": ["BLOCKER", "MAJOR", "MINOR", "NIT"]}, "file": {"type": "string"}, "line": {"type": ["number", "null"]},
        "rule": {"type": "string"}, "message": {"type": "string"},
        "repro": {"type": "object", "properties": {"command": {"type": "string"}, "failing": {"type": "boolean"}}}}}}}}


def wave_section(text, n):
    lines = text.splitlines()
    start = end = None
    for i, line in enumerate(lines):
        m = re.match(r"^#{1,6}\s*wave\s*(\d+)\b", line, re.I)
        if m and int(m.group(1)) == n:
            start = i
        elif m and start is not None:
            end = i
            break
    return "\n".join(lines[start:end]) if start is not None else ""


def base_text(rel):
    try:
        return subprocess.run(["git", "-C", MAIN, "show", f"{BASE}:{rel}"], capture_output=True, text=True, encoding="utf-8",
                              errors="replace", check=True).stdout
    except subprocess.CalledProcessError:
        return ""


def product_diff():
    out = []
    for rel in PRODUCT:
        cur_path = WT / rel
        if not cur_path.exists():
            continue
        cur = cur_path.read_text(encoding="utf-8", errors="replace").replace("\r\n", "\n")
        old = base_text(rel).replace("\r\n", "\n")
        if cur == old:
            continue
        diff = difflib.unified_diff(old.splitlines(), cur.splitlines(), f"a/{rel}", f"b/{rel}", lineterm="", n=4)
        out.append("\n".join(diff))
    return "\n\n".join(out) if out else "(no product file differs from the base commit)"


def gate_summary(n):
    try:
        g = json.loads((WT / ".foreman" / f"wave-{n}-gate.json").read_text(encoding="utf-8"))
    except (OSError, ValueError):
        return "(no gate artifact yet)"
    text = (g.get("stdout") or "") + "\n" + (g.get("stderr") or "")
    events = [l.strip()[:200] for l in text.splitlines() if "::" in l and re.search(r"\b(PASSED|FAILED|ERROR|XFAIL|XPASS|SKIPPED)\b", l)]
    summary = [l for l in text.splitlines() if re.search(r"=+ .*(passed|failed).* =+", l)]
    return f"exit {g.get('exit_code')} · green {g.get('green')} · counts {g.get('tap')}\n" + "\n".join(summary[-1:]) + "\n" + "\n".join(events)


def numbered(rel):
    text = (WT / rel).read_text(encoding="utf-8", errors="replace").replace("\r\n", "\n")
    return "\n".join(f"{i + 1:5d}| {line}" for i, line in enumerate(text.splitlines()))


def main(wave, seat, out_path):
    wave = int(wave)
    plan = PLAN.read_text(encoding="utf-8")
    parts = [
        "READ-ONLY SEAT (headless grok): every file you need is supplied below with line numbers. Do NOT run shell commands "
        "(a command needing approval cancels your turn unanswered). If you must look at something else, use only your "
        "file-read tool. Always finish with the JSON you were asked for.",
        "",
        f"You are an INDEPENDENT, READ-ONLY reviewer (#{seat}) for Foreman wave {wave} of Anchor's provider-integration "
        "assurance. Your job is to REFUTE, not to praise. The ground-truth gate was run by the orchestrator; its artifact "
        "summary is below. Do NOT trust any pasted command output in prose. Cite file:line (from the numbered listings) or a "
        "failing repro command+output for every finding. First answer answerable-from-frozen-docs: yes/no with a cited plan line.",
        "",
        "Review this wave's work against the frozen contract: (1) do the required tests establish the behaviours the plan "
        "names, at the actual mechanism (real server, cookie auth, substituted native commands), or do they mirror an "
        "implementation / assert something vacuous? (2) are the strict-xfail reproductions honest failing reproductions of "
        "real defects? (3) are there defects in the provider integration itself — credential exposure, account separation, "
        "failure paths (timeouts, cancellation, corrupt settings, stale/unknown/exhausted allowance), concurrency, routing — "
        "that neither the existing tests nor the new ones catch? (4) was any existing assertion weakened? Severity: BLOCKER "
        "= would be wrong to ship / credential or identity hazard; MAJOR = a real defect or a required behaviour left "
        "unproved; MINOR = honesty/robustness gap; NIT = style. Every finding needs file:line evidence or a repro command.",
        "",
        "===== FROZEN PLAN — THIS WAVE'S CONTRACT =====",
        wave_section(plan, wave),
        "",
        f"===== ORCHESTRATOR GATE ARTIFACT (wave {wave}, latest iteration) =====",
        gate_summary(wave),
        "",
        "===== PRODUCT DIFF vs the worktree's base commit f77710c (the author's landed work is the base; this is what the "
        "assurance waves changed) =====",
        product_diff(),
    ]
    for rel in TESTS[wave]:
        if (WT / rel).exists():
            parts += ["", f"===== {rel} (new in this assurance; numbered) =====", numbered(rel)]
    for rel in INLINE_FULL[wave]:
        parts += ["", f"===== {rel} (current, numbered) =====", numbered(rel)]
    parts += ["", "Respond with ONLY a single raw JSON object (no markdown fences, no prose) that conforms to this JSON Schema:",
              json.dumps(SCHEMA)]
    Path(out_path).write_text("\n".join(parts), encoding="utf-8")
    print(out_path, len("\n".join(parts)), "chars")


if __name__ == "__main__":
    main(*sys.argv[1:4])
